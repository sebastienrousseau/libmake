10/// Constructs the command-line interface and parses the arguments this
11/// process was started with.
12///
13/// This is a thin wrapper over [`build_from`] that reads
14/// [`std::env::args_os`]. Prefer [`build_from`] anywhere the arguments
15/// should not depend on how the process was launched — tests and
16/// doctests especially, since there the arguments belong to the test
17/// harness rather than to this CLI.
18///
19/// # Errors
20///
21/// Returns an error if parsing the process arguments fails.
22pub fn build() -> Result<ArgMatches, Error> {
23    build_from(std::env::args_os())
24}
25
26/// Constructs the command-line interface and parses `args`.
27///
28/// `args` follows the usual convention: the first element is the
29/// program name and is ignored for matching purposes.
30///
31/// # Examples
32///
33/// ```
34/// use libmake::cli;
35///
36/// // Deterministic: does not inherit the caller's arguments.
37/// let matches = cli::build_from(["libmake"]).expect("CLI parsing failed");
38/// assert!(matches.subcommand_name().is_none());
39/// ```
40///
41/// # Errors
42///
43/// Returns an error if the supplied arguments fail to parse.
44pub fn build_from<I, T>(args: I) -> Result<ArgMatches, Error>
45where
46    I: IntoIterator<Item = T>,
47    T: Into<std::ffi::OsString> + Clone,
48{
49    let manual_args = vec![
50            create_arg_info("author", Some("Me"), "Sets the author of the library", 'a', "author", "AUTHOR"),
51            create_arg_info("build", Some("build.rs"), "Sets the build script that is used to perform additional build-time operations.", 'b', "build", "BUILD"),
52            create_arg_info("categories", Some("['category 1', 'category 2']"), "Sets the categories of the library", 'c', "categories", "CATEGORIES"),
53            create_arg_info("description", Some("A library for doing things"), "Sets the description of the library", 'd', "description", "DESCRIPTION"),
54            create_arg_info("documentation", Some("https://lib.rs/crates/my_library"), "Sets the documentation URL of the library", 'u', "documentation", "DOCUMENTATION"),
55            create_arg_info("edition", Some("2021"), "Sets the edition of the library", 'e', "edition", "EDITION"),
56            create_arg_info("email", Some("test@test.com"), "Sets the email of the library author", '@', "email", "EMAIL"),
57            create_arg_info("homepage", Some("https://test.com"), "Sets the homepage of the library", 'p', "homepage", "HOMEPAGE"),
58            create_arg_info("keywords", Some("['keyword1', 'keyword2']"), "Sets the keywords of the library", 'k', "keywords", "KEYWORDS"),
59            create_arg_info("license", Some("MIT OR Apache-2.0"), "Sets the license of the library", 'l', "license", "LICENSE"),
60            create_arg_info("name", Some("my_library"), "Sets the name of the library", 'n', "name", "NAME"),
61            create_arg_info("output", Some("my_library"), "Sets the output directory for the library", 'o', "output", "OUTPUT"),
62            create_arg_info("readme", Some("README.md"), "Sets the README file for the library", 'm', "readme", "README"),
63            create_arg_info("repository", Some("https://github.com/example/my_library"), "Sets the repository URL of the library", 'g', "repository", "REPOSITORY"),
64            create_arg_info("rustversion", Some("1.60"), "Sets the Rust version of the library", 'r', "rustversion", "RUSTVERSION"),
65            create_arg_info("version", Some("0.2.6"), "Sets the version of the library", 'v', "version", "VERSION"),
66            create_arg_info("website", Some("https://test.com"), "Sets the website of the library author", 'w', "website", "WEBSITE"),
67        ];
68
69    let file_args = vec![
70        create_arg_info(
71            "csv",
72            Some(""),
73            "Sets the CSV file to use for generating the library",
74            'x',
75            "csv",
76            "CSV",
77        ),
78        create_arg_info(
79            "ini",
80            Some(""),
81            "Sets the INI file to use for generating the library",
82            'i',
83            "ini",
84            "INI",
85        ),
86        create_arg_info(
87            "json",
88            Some(""),
89            "Sets the JSON file to use for generating the library",
90            'j',
91            "json",
92            "JSON",
93        ),
94        create_arg_info(
95            "yaml",
96            Some(""),
97            "Sets the YAML file to use for generating the library",
98            'y',
99            "yaml",
100            "YAML",
101        ),
102        create_arg_info(
103            "toml",
104            Some(""),
105            "Sets the TOML file to use for generating the library",
106            't',
107            "toml",
108            "TOML",
109        ),
110    ];
111
112    let command = Command::new("My Library")
113        .author("Sebastien Rousseau")
114        .about("A Rust library generator that helps create high-quality Rust libraries quickly and easily.")
115        .after_help("By default, if no arguments are passed in, the CLI will throw an error. To see a list of available actions, run `--help`.")
116        .subcommand(
117            Command::new("manual")
118                .about("Set library information manually")
119                .args(manual_args.into_iter().map(create_arg).collect::<Vec<Arg>>()),
120        )
121        .subcommand(
122            Command::new("file")
123                .about("Set library information from a file")
124                .args(file_args.into_iter().map(create_arg).collect::<Vec<Arg>>()),
125        );
126
127    // Assuming validate_args is a custom function that you have implemented
128    let matches = command.clone().try_get_matches_from(args)?;
129
130    Ok(matches)
131}

133/// Helper function to create a command-line argument.
134const fn create_arg_info(
135    name: &'static str,
136    default: Option<&'static str>,
137    help: &'static str,
138    short: char,
139    long: &'static str,
140    value_name: &'static str,
141) -> (
142    &'static str,
143    Option<&'static str>,
144    &'static str,
145    char,
146    &'static str,
147    &'static str,
148) {
149    (name, default, help, short, long, value_name)
150}
151
152/// Creates an argument based on provided information.
153pub fn create_arg(
154    arg_info: (
155        &'static str,
156        Option<&'static str>,
157        &'static str,
158        char,
159        &'static str,
160        &'static str,
161    ),
162) -> Arg {
163    let (name, default, help, short, long, value_name) = arg_info;
164
165    let mut arg = Arg::new(name)
166        .help(help)
167        .short(short)
168        .long(long)
169        .value_name(value_name);
170
171    if let Some(default_value) = default {
172        arg = arg.default_value(default_value);
173    }
174    arg
175}
