14/// Reads a file and deserializes its content using the specified deserializer function.
15///
16/// # Arguments
17///
18/// * `file_path` - The path of the file to read.
19/// * `deserializer` - A function that takes a `File` and returns a deserialized value of type `T`.
20///
21/// # Returns
22///
23/// Returns a `Result<T, Box<dyn std::error::Error>>` containing the deserialized value, or an error if one occurs.
24///
25fn read_file<T, F>(
26    file_path: &Path,
27    deserializer: F,
28) -> Result<T, Box<dyn std::error::Error>>
29where
30    F: FnOnce(File) -> Result<T, Box<dyn std::error::Error>>,
31{
32    let file = File::open(file_path)?;
33    deserializer(file)
34}

36/// Reads a CSV file at the given file path and returns the value of
37/// the given field.
38///
39/// # Arguments
40///
41/// * `file_path` - An optional string slice that holds the file path of the CSV file to read.
42/// * `field_index` - The index of the field to retrieve.
43///
44pub fn get_csv_field(
45    file_path: Option<&str>,
46    field_index: usize,
47) -> Option<Vec<String>> {
48    file_path.and_then(|file_path| {
49        let current_dir = env::current_dir().ok()?;
50        let file_path = Path::new(&current_dir).join(file_path);
51        let file = File::open(file_path).ok()?;
52        let mut rdr = csv::Reader::from_reader(file);
53
54        let mut values = Vec::new();
55        for result in rdr.records() {
56            let record = result.ok()?;
57            // `?` propagates the out-of-range field index as None.
58            values.push(record.get(field_index)?.to_string());
59        }
60        if values.is_empty() {
61            None
62        } else {
63            Some(values)
64        }
65    })
66}

72/// Retrieves a specific field's value from a configuration file.
73///
74/// # Arguments
75///
76/// * `file_path` - An optional reference to the path of the configuration file.
77/// * `file_format` - The format of the configuration file ("json", "yaml", or "ini").
78/// * `field_name` - The name of the field to retrieve the value from.
79///
80/// # Returns
81///
82/// Returns a `Result<String, Box<dyn std::error::Error>>` containing the value of the specified field, or an error if one occurs.
83///
84pub fn get_config_field(
85    file_path: Option<&str>,
86    file_format: Option<&str>,
87    field_name: &str,
88) -> Result<String, Box<dyn std::error::Error>> {
89    // Ensure file_path is provided
90    let file_path = file_path.ok_or("File path is not provided")?;
91
92    // Ensure file_format is provided and is either 'json', 'yaml', or 'ini'
93    let format = file_format.ok_or("File format is not provided")?;
94    match format {
95        "ini" => get_ini_field(Some(file_path), field_name),
96        "json" => get_json_field(Some(file_path), field_name),
97        "yaml" => get_yaml_field(Some(file_path), field_name),
98        _ => Err(format!(
99            "Unsupported file format: {}. Supported formats are 'json', 'yaml', and 'ini'.",
100            format
101        )
102        .into()),
103    }
104}

106/// Creates a directory at the specified path.
107///
108/// # Arguments
109///
110/// * `path` - The path where the directory should be created.
111///
112/// # Errors
113///
114/// Returns an `io::Error` if the directory cannot be created. This could be due to
115/// various reasons such as insufficient permissions, the directory already existing,
116/// or other I/O-related errors.
117///
118pub fn create_directory(path: &Path) -> io::Result<()> {
119    fs::create_dir(path).or_else(|e| match e.kind() {
120        io::ErrorKind::AlreadyExists => Ok(()),
121        _ => Err(e),
122    })
