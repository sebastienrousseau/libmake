15/// Creates the template folder and downloads necessary template files.
16///
17/// This function attempts to create a template directory in the current working directory
18/// and then downloads a set of predefined template files into this directory.
19///
20/// # Errors
21///
22/// Returns an `io::Error` if the template directory cannot be created, or if there's an error
23/// during the download and writing of the template files. Possible causes include
24/// network issues, file permission errors, or other I/O-related problems.
25///
26pub fn create_template_folder() -> io::Result<()> {
27    let current_dir = std::env::current_dir()?;
28    // println!("Current directory: {:?}", current_dir);
29    let template_dir_path = current_dir.join("template");
30    // println!("Creating template directory: {:?}", template_dir_path);
31    create_directory(&template_dir_path)?;
32    let url = "https://raw.githubusercontent.com/sebastienrousseau/libmake/main/template/";
33    let files = [
34        "AUTHORS.tpl",
35        "build.tpl",
36        "Cargo.tpl",
37        "ci.tpl",
38        "CONTRIBUTING.tpl",
39        "criterion.tpl",
40        "deepsource.tpl",
41        "deny.tpl",
42        "example.tpl",
43        "gitignore.tpl",
44        "lib.tpl",
45        "macros.tpl",
46        "main.tpl",
47        "README.tpl",
48        "rustfmt.tpl",
49        "TEMPLATE.tpl",
50        "test.tpl",
51    ];
52    for file in &files {
53        let file_path = template_dir_path.join(file);
54        // Check if the file already exists
55        if !file_path.exists() {
56            let file_url = format!("{url}{file}");
57            let response =
58                reqwest::blocking::get(&file_url).map_err(|e| {
59                    io::Error::other(format!(
60                        "Failed to download template file: {e}"
61                    ))
62                })?;
63
64            let file_contents = response.text().map_err(|e| {
65                io::Error::other(format!(
66                    "Failed to read response body: {e}"
67                ))
68            })?;
69
70            // Write the file contents, trimming any leading or trailing newline characters
71            fs::write(
72                &file_path,
73                file_contents
74                    .trim_start_matches('\n')
75                    .trim_end_matches('\n'),
76            )?;
77        }
78    }
79    Ok(())
80}

82/// Copies a template file to the output directory and replaces the
83/// placeholders with the given parameters (if any).
84///
85/// # Arguments
86///
87/// * `template_file` - The name of the template file.
88/// * `dest_file` - The name of the destination file.
89/// * `project_directory` - The path to the project directory.
90/// * `params` - The parameters to be used for replacing the placeholders.
91///
92/// # Errors
93///
94/// Returns an `io::Error` in the following cases:
95///
96/// - If the template file cannot be found, read, or copied.
97/// - If the destination file cannot be created or written to.
98/// - If there is an error in replacing placeholders in the destination file.
99///
100#[allow(unused_results)]
101pub fn copy_and_replace_template(
102    template_file: &str,
103    dest_file: &str,
104    project_directory: &PathBuf,
105    params: &FileGenerationParams,
106) -> io::Result<()> {
107    let mut tpl_file = PathBuf::new();
108    tpl_file.push("template");
109    tpl_file.push(template_file);
110
111    let mut dest_path = PathBuf::new();
112    dest_path.push(project_directory);
113    dest_path.push(dest_file);
114
115    fs::copy(&tpl_file, &dest_path)?;
116
117    replace_placeholders(&tpl_file, &dest_path, params)?;
118
119    Ok(())
120}

122/// Generates files for a new Rust project based on given arguments.
123///
124/// # Arguments
125///
126/// - `params` - Parameters containing project details and configurations.
127///
128/// # Errors
129///
130/// Returns an `io::Result` error if:
131///
132/// - There are issues creating the project directory or subdirectories.
133/// - There are issues copying or replacing the template files.
134/// - Any other I/O related errors occur during the file generation process.
135///
136pub fn generate_files(params: FileGenerationParams) -> io::Result<()> {
137    let Some(ref output) = params.output else {
138        return Err(io::Error::other(
139            "Output directory is not specified",
140        ));
141    };
142
143    // Get the project directory path from the output parameter,
144    // create a PathBuf from it, and assign it to the project_directory variable
145    let project_directory =
146        PathBuf::from(output.clone().trim_matches('\"'));
147
148    // Creating the project directory
149    create_directory(&project_directory)?;
150
151    // Creating the template directory
152    create_template_folder()?;
153
154    // Define the subdirectories to be created within the project directory
155    let subdirectories = [
156        "src",
157        "benches",
158        "examples",
159        "tests",
160        ".github/",
161        ".github/workflows",
162    ];
163
164    // Iterate over the subdirectories and create them
165    for subdir in &subdirectories {
166        let dir_path = project_directory.join(subdir);
167        create_directory(&dir_path)?;
168    }
169
170    // Copying the template files to the new library directory
171    let templates = [
172        // --- #Start GitHub Actions workflows ---
173        // Add audit GitHub Actions workflows template
174        ("github/workflows/audit.tpl", ".github/workflows/audit.yml"),
175        // Add check GitHub Actions workflows template
176        ("github/workflows/check.tpl", ".github/workflows/check.yml"),
177        // Add coverage GitHub Actions workflows template
178        (
179            "github/workflows/coverage.tpl",
180            ".github/workflows/coverage.yml",
181        ),
182        // Add document GitHub Actions workflows template
183        (
184            "github/workflows/document.tpl",
185            ".github/workflows/document.yml",
186        ),
187        // Add lint GitHub Actions workflows template
188        ("github/workflows/lint.tpl", ".github/workflows/lint.yml"),
189        // Add release GitHub Actions workflows template
190        (
191            "github/workflows/release.tpl",
192            ".github/workflows/release.yml",
193        ),
194        // Add test GitHub Actions workflows template
195        ("github/workflows/test.tpl", ".github/workflows/test.yml"),
196        // --- #End GitHub Actions workflows ---
197        // Add Authors template
198        ("AUTHORS.tpl", "AUTHORS.md"),
199        // Add build template
200        ("build.tpl", "build.rs"),
201        // Add Cargo template
202        ("Cargo.tpl", "Cargo.toml"),
203        // Add Contributing template
204        ("CONTRIBUTING.tpl", "CONTRIBUTING.md"),
205        // Add Criterion template
206        ("criterion.tpl", "benches/criterion.rs"),
207        // Add Deepsource template
208        ("deepsource.tpl", ".deepsource.toml"),
209        // Add Deny template
210        ("deny.tpl", "deny.toml"),
211        // Add Example template
212        ("example.tpl", "examples/example.rs"),
213        // Add Gitignore template
214        ("gitignore.tpl", ".gitignore"),
215        // Add Lib template
216        ("lib.tpl", "src/lib.rs"),
217        // Add Macros template
218        ("macros.tpl", "src/macros.rs"),
219        // Add Main template
220        ("main.tpl", "src/main.rs"),
221        // Add Readme template
222        ("README.tpl", "README.md"),
223        // Add Rustfmt template
224        ("rustfmt.tpl", "rustfmt.toml"),
225        // Add Template template
226        ("TEMPLATE.tpl", "TEMPLATE.md"),
227        // Add Test template
228        ("test_test.tpl", "tests/test_test.rs"),
229    ];
230
231    for (template, target) in templates {
232        copy_and_replace_template(
233            template,
234            target,
235            &project_directory,
236            &params,
237        )?;
238    }
239
240    // Displaying the argument and value pairs
241    println!("{:<15}Value", "Argument");
242    println!("{:<15}{}", "author", params.author.unwrap_or_default());
243    println!("{:<15}{}", "build", params.build.unwrap_or_default());
244    println!(
245        "{:<15}{}",
246        "categories",
247        params.categories.unwrap_or_default()
248    );
249    println!(
250        "{:<15}{}",
251        "description",
252        params.description.unwrap_or_default()
253    );
254    println!(
255        "{:<15}{}",
256        "documentation",
257        params.documentation.unwrap_or_default()
258    );
259    println!("{:<15}{}", "edition", params.edition.unwrap_or_default());
260    println!("{:<15}{}", "email", params.email.unwrap_or_default());
261    println!(
262        "{:<15}{}",
263        "homepage",
264        params.homepage.unwrap_or_default()
265    );
266    println!(
267        "{:<15}{}",
268        "keywords",
269        params.keywords.unwrap_or_default()
270    );
271    println!("{:<15}{}", "license", params.license.unwrap_or_default());
272    println!("{:<15}{}", "name", params.name.unwrap_or_default());
273    println!("{:<15}{}", "output", output.clone());
274    println!("{:<15}{}", "readme", params.readme.unwrap_or_default());
275    println!(
276        "{:<15}{}",
277        "repository",

291/// Generates files for a new Rust project based on a configuration file.
292///
293/// # Arguments
294///
295/// - `path` - The path to the configuration file.
296/// - `file_type` - The type of the configuration file (e.g., JSON, YAML, CSV, TOML and INI).
297///
298/// # Errors
299///
300/// Returns an `io::Result` error if:
301///
302/// - The specified configuration file cannot be found, read, or is not in a valid format.
303/// - There are issues parsing the configuration file into the `FileGenerationParams` struct.
304/// - Any errors occur during the file generation process based on the configuration.
305///
306pub fn generate_from_config(
307    path: &str,
308    file_type: &str,
309) -> Result<(), String> {
310    match file_type {
311        "csv" => macro_generate_from_csv!(path)?,
312        "ini" => macro_generate_from_ini!(path)?,
313        "json" => macro_generate_from_json!(path)?,
314        "yaml" => macro_generate_from_yaml!(path)?,
315        "toml" => macro_generate_from_toml!(path)?,
316        _ => {
317            return Err(format!(
318                "Unsupported configuration file type: {}",
319                file_type
320            ))
321        }
322    }
323    Ok(())
324}
