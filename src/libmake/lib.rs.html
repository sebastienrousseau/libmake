24//! ## Overview
25//!
26//! `LibMake` is a tool designed to quickly help create high-quality
27//! Rust libraries by generating a set of pre-filled and pre-defined
28//! templated files. This opinionated boilerplate scaffolding tool aims
29//! to greatly reduce development time and minimize repetitive tasks,
30//! allowing you to focus on your business logic while enforcing
31//! standards, best practices, consistency, and providing style guides
32//! for your library.
33//!
34//! With `LibMake`, you can easily generate a new Rust library code base
35//! structure with all the necessary files, layouts, build
36//! configurations, code, tests, benchmarks, documentation, and much
37//! more in a matter of seconds.
38//!
39//! The library is designed to be used as a command-line tool. It is
40//! available on [Crates.io](https://crates.io/crates/libmake) and
41//! [Lib.rs](https://lib.rs/crates/libmake).
42//!
43//! ## Features
44//!
45//! `LibMake` offers the following features and benefits:
46//!
47//! - Create your Rust library with ease using the command-line
48//!   interface or by providing a configuration file in CSV, JSON, or
49//!   YAML format.
50//! - Rapidly generate new library projects with a pre-defined structure
51//!   and boilerplate code that you can customize with your own template.
52//! - Automatically generate basic functions, methods, and macros to get
53//!   you started with your Rust library.
54//! - Enforce best practices and standards with starter documentation,
55//!   test suites, and benchmark suites that are designed to help you
56//!   get up and running quickly.

112/// Initializes the logger with a file logger and a terminal logger and processes
113/// command-line arguments to generate the new library.
114///
115/// # Errors
116///
117/// This function will return an error in the following situations:
118///
119/// - If there is a problem generating ASCII art for the tool's CLI.
120/// - If an error occurs while building the command-line interface or processing arguments.
121/// - Any other errors that may arise from operations performed within the function.
122///
123pub fn run() -> Result<(), Box<dyn Error>> {
124    let date = DateTime::new();
125    // dtt 0.0.11 made the struct fields pub(crate); the public
126    // accessor is format_rfc3339(), which is ISO 8601 compatible.
127    let iso = date.format_rfc3339()?;
128
129    // Open the log file for appending
130    let mut log_file = File::create("./libmake.log")?;
131
132    // Generate ASCII art for the tool's CLI
133    let log = Log {
134        format: LogFormat::CLF,
135        ..Log::build(
136            LogLevel::INFO,
137            "ASCII art generation event started.",
138        )
139        .time(&iso)
140        .component("deps")
141    };
142    // Write the log to both the console and the file
143    writeln!(log_file, "{}", log)?;
144
145    // Printing the ASCII art to the console
146    println!("{}", macro_ascii!("LibMake"));
147
148    let log = Log {
149        format: LogFormat::CLF,
150        ..Log::build(
151            LogLevel::INFO,
152            "ASCII art generation event completed.",
153        )
154        .time(&iso)
155        .component("deps")
156    };
157    // Write the log to both the console and the file
158    writeln!(log_file, "{}", log)?;
159
160    // Build the command-line interface and process the arguments
161    let matches = build()?;
162    process_arguments(&matches)?;
163
164    // Check the number of arguments, provide a welcome message if no arguments were passed
165    // The old `macro_log!` emitted as a side effect; `Log::build` only
166    // constructs, so this entry has to be emitted explicitly or it is
167    // silently dropped.
168    Log {
169        format: LogFormat::CLF,
170        ..Log::build(
171            LogLevel::INFO,
172            "Welcome to LibMake! 👋\n\nLet's get started! Please, run `libmake --help` for more information.",
173        )
174        .time(&iso)
175        .component("cli")
176    }
177    .log();
178    eprintln!(
179        "\n\nWelcome to LibMake! 👋\n\nLet's get started! Please, run `libmake --help` for more information.\n"
180    );
181
182    Ok(())
183}
